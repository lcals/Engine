cmake_minimum_required(VERSION 3.25)
project(engine_performance_tools LANGUAGES C CXX)

# 独立工具构建：不进入引擎构建，也不下载或重编引擎依赖。
set(TRACY_SOURCE_DIR "" CACHE PATH "Existing Tracy 0.14.1 source directory")
set(PERFORMANCE_TOOL "analyze" CACHE STRING "Build capture or analyze tools")
set_property(CACHE PERFORMANCE_TOOL PROPERTY STRINGS capture analyze)
if(NOT EXISTS "${TRACY_SOURCE_DIR}/public/common/TracyVersion.hpp")
    message(FATAL_ERROR "TRACY_SOURCE_DIR must point to existing Tracy 0.14.1 sources")
endif()
include("${TRACY_SOURCE_DIR}/cmake/version.cmake")
if(NOT TRACY_VERSION_STRING VERSION_EQUAL "0.14.1")
    message(FATAL_ERROR "These offline tools require Tracy 0.14.1")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_FIND_PACKAGE_TARGETS_GLOBAL ON)
# 只接受 vcpkg 已准备的离线工具源码；未补丁的上游 vendor.cmake 会隐式下载。
file(READ "${TRACY_SOURCE_DIR}/cmake/vendor.cmake" PERFORMANCE_TRACY_VENDOR)
if(PERFORMANCE_TRACY_VENDOR MATCHES "CPMAddPackage|FetchContent")
    message(FATAL_ERROR "Use vcpkg-prepared Tracy sources without implicit vendor downloads")
endif()
find_package(Threads REQUIRED)
find_package(capstone CONFIG REQUIRED)
find_package(zstd CONFIG REQUIRED)
find_package(PPQSort CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
set(CMAKE_DISABLE_FIND_PACKAGE_Git ON)
set(NO_LTO ON)
set(NO_ISA_EXTENSIONS ON)
set(NO_CCACHE ON)
if(WIN32)
    find_program(PERFORMANCE_COMPILER_CACHE sccache REQUIRED)
else()
    find_program(PERFORMANCE_COMPILER_CACHE ccache REQUIRED)
endif()
set(CMAKE_C_COMPILER_LAUNCHER "${PERFORMANCE_COMPILER_CACHE}")
set(CMAKE_CXX_COMPILER_LAUNCHER "${PERFORMANCE_COMPILER_CACHE}")

# 两个独立配置保留上游 capture 的 NO_STATISTICS 和 csvexport 的统计版本，
# 避免为实时采集进程额外计算 zone 统计。依赖使用已有 vcpkg 前缀。
if(PERFORMANCE_TOOL STREQUAL "capture")
    add_subdirectory("${TRACY_SOURCE_DIR}/capture" capture)
elseif(PERFORMANCE_TOOL STREQUAL "analyze")
    add_subdirectory("${TRACY_SOURCE_DIR}/csvexport" csvexport)
    add_executable(tracy-frameexport frameexport.cpp)
    target_link_libraries(tracy-frameexport PRIVATE TracyServer)
    if(MSVC)
        target_compile_options(tracy-frameexport PRIVATE /utf-8)
    endif()
else()
    message(FATAL_ERROR "PERFORMANCE_TOOL must be capture or analyze")
endif()
